  <script>

    const form = document.getElementById("carbonForm");

    const result = document.getElementById("result");

    const scoreElement = document.getElementById("score");

    const levelElement = document.getElementById("level");

    const analysisText =
      document.getElementById("analysisText");

    const meterFill =
      document.getElementById("meterFill");

    const recommendationsList =
      document.getElementById("recommendationsList");


    form.addEventListener("submit", function(event) {

      event.preventDefault();


      // ==========================
      // OBTENER LOS PUNTAJES
      // ==========================

      const transport =
        Number(document.getElementById("transport").value);

      const distance =
        Number(document.getElementById("distance").value);

      const meat =
        Number(document.getElementById("meat").value);

      const local =
        Number(document.getElementById("local").value);

      const energy =
        Number(document.getElementById("energy").value);

      const waste =
        Number(document.getElementById("waste").value);

      const consumption =
        Number(document.getElementById("consumption").value);

      const water =
        Number(document.getElementById("water").value);


      // ==========================
      // CALCULAR PUNTAJE
      // ==========================

      const total =
        transport +
        distance +
        meat +
        local +
        energy +
        waste +
        consumption +
        water;


      const maximumScore = 48;

      const percentage =
        (total / maximumScore) * 100;


      // ==========================
      // MOSTRAR PUNTAJE
      // ==========================

      scoreElement.textContent = total;


      // ==========================
      // DETERMINAR NIVEL
      // ==========================

      let level = "";

      let analysis = "";

      let className = "";

      if (total <= 16) {

        level = "🟢 Huella baja";

        analysis =
          "Tus hábitos evaluados muestran una presión ambiental relativamente baja. " +
          "Mantienes buenas prácticas relacionadas con movilidad, consumo de recursos " +
          "y manejo de residuos. El reto es mantener estos hábitos y continuar buscando " +
          "pequeñas acciones que permitan reducir aún más tu impacto.";

        className = "low";

      }

      else if (total <= 28) {

        level = "🟡 Huella moderada";

        analysis =
          "Tu resultado muestra un nivel intermedio de impacto ambiental. " +
          "Tienes algunos hábitos positivos, pero también existen oportunidades " +
          "concretas de mejora. Prestar especial atención a la movilidad, alimentación, " +
          "energía y consumo puede generar una reducción importante de tu impacto.";

        className = "moderate";

      }

      else if (total <= 38) {

        level = "🟠 Huella alta";

        analysis =
          "Tus hábitos evaluados representan una presión ambiental relativamente alta. " +
          "No es necesario cambiar todo al mismo tiempo. Lo más efectivo es identificar " +
          "las categorías con mayor puntaje y comenzar por cambios sencillos y sostenibles.";

        className = "high";

      }

      else {

        level = "🔴 Huella muy alta";

        analysis =
          "El resultado muestra una presión ambiental elevada dentro de esta escala. " +
          "Existen varias oportunidades para reducir tu impacto. Te recomendamos comenzar " +
          "por los hábitos relacionados con movilidad, alimentación, consumo de energía " +
          "y generación de residuos.";

        className = "very-high";

      }


      levelElement.textContent = level;

      analysisText.textContent = analysis;


      // ==========================
      // BARRA DE RESULTADO
      // ==========================

      meterFill.style.width =
        percentage + "%";

      meterFill.className =
        "meter-fill " + className;


      // ==========================
      // MOSTRAR CATEGORÍAS
      // ==========================

      document.getElementById(
        "transportScore"
      ).textContent =
        transport + distance;


      document.getElementById(
        "foodScore"
      ).textContent =
        meat + local;


      document.getElementById(
        "energyScore"
      ).textContent =
        energy;


      document.getElementById(
        "wasteScore"
      ).textContent =
        waste;


      document.getElementById(
        "consumptionScore"
      ).textContent =
        consumption;


      document.getElementById(
        "waterScore"
      ).textContent =
        water;


      // ==========================
      // RECOMENDACIONES
      // ==========================

      const recommendations = [];


      if (transport + distance >= 8) {

        recommendations.push(
          "Prioriza caminar, usar bicicleta o transporte público cuando sea posible. También puedes compartir el vehículo para reducir viajes individuales."
        );

      }


      if (meat >= 4) {

        recommendations.push(
          "Reduce progresivamente el consumo de carne de res y alterna con opciones de menor impacto ambiental."
        );

      }


      if (local >= 4) {

        recommendations.push(
          "Prefiere alimentos locales, de temporada y producidos cerca de tu lugar de consumo."
        );

      }


      if (energy >= 4) {

        recommendations.push(
          "Apaga luces y equipos que no estés utilizando y desconecta cargadores y aparatos cuando sea posible."
        );

      }


      if (waste >= 4) {

        recommendations.push(
          "Separa los residuos desde el lugar donde se generan y aprovecha los materiales reciclables."
        );

      }


      if (consumption >= 4) {

        recommendations.push(
          "Antes de comprar, pregúntate si realmente necesitas el producto. Prioriza reparar, reutilizar y comprar productos durables."
        );

      }


      if (water >= 4) {

        recommendations.push(
          "Reduce el tiempo de ducha, evita desperdicios y revisa posibles fugas de agua en casa."
        );

      }


      if (recommendations.length === 0) {

        recommendations.push(
          "¡Excelente! Mantén tus hábitos sostenibles y compártelos con familiares, compañeros y amigos."
        );

      }


      recommendationsList.innerHTML = "";


      recommendations.forEach(function(recommendation) {

        const li =
          document.createElement("li");

        li.textContent =
          recommendation;

        recommendationsList.appendChild(li);

      });


      // ==========================
      // MOSTRAR RESULTADO
      // ==========================

      result.style.display = "block";


      result.scrollIntoView({
        behavior: "smooth",
        block: "start"
      });

    });


    // ==========================
    // BOTÓN REINICIAR
    // ==========================

    document
      .getElementById("resetButton")
      .addEventListener("click", function() {

        form.reset();

        result.style.display = "none";

        meterFill.style.width = "0%";

        window.scrollTo({
          top: 0,
          behavior: "smooth"
        });

      });

  </script>
